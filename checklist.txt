### 1.2 Data Models (Mongoose/Prisma Schemas)

#### **Session Model**
```typescript
□ SessionSchema
  Fields:
  - sessionId: String (unique, indexed)
  - userId: String (indexed)
  - subject: String
  - createdAt: Date
  - updatedAt: Date
  - status: Enum ['active', 'completed', 'archived']
  
  - narrativeContext: {
      mainTheme: String
      realWorldProblem: String
      targetAudience: String
      learningObjectives: [String]
    }
  
  - metadata: {
      difficulty: Enum ['beginner', 'intermediate', 'advanced']
      estimatedDuration: Number
      language: String
    }
  
  - topicIds: [ObjectId] (ref: 'Topic')
  - currentTopicId: ObjectId (ref: 'Topic')
  - currentSubtopicId: ObjectId (ref: 'Subtopic')
  
  Methods:
  - generateTopics()
  - getCurrentProgress()
  - getNextSubtopic()
```

#### **Topic Model**
```typescript
□ TopicSchema
  Fields:
  - topicId: String (unique)
  - sessionId: ObjectId (ref: 'Session', indexed)
  - title: String
  - description: String
  - order: Number
  - createdAt: Date
  
  - prerequisites: [ObjectId] (ref: 'Topic')
  - dependents: [ObjectId] (ref: 'Topic') // Topics that depend on this
  - subtopicIds: [ObjectId] (ref: 'Subtopic')
  
  - narrativeRole: String // How this fits in the overall story
  - keyQuestions: [String] // Questions this topic answers
  
  - generationStatus: Enum ['pending', 'generating', 'completed', 'failed']
  - generatedAt: Date
  
  Methods:
  - generateSubtopics()
  - checkPrerequisites()
  - getNextTopic()
```

#### **Subtopic Model**
```typescript
□ SubtopicSchema
  Fields:
  - subtopicId: String (unique)
  - topicId: ObjectId (ref: 'Topic', indexed)
  - title: String
  - description: String
  - order: Number
  - createdAt: Date
  
  - prerequisites: [ObjectId] (ref: 'Subtopic')
  - estimatedTime: Number // in minutes
  
  - contentId: ObjectId (ref: 'Content')
  - generationStatus: Enum ['pending', 'generating', 'completed', 'failed']
  - generatedAt: Date
  
  Methods:
  - generateContent()
  - isAvailable() // Check if prerequisites are met
```

#### **Content Model**
```typescript
□ ContentSchema
  Fields:
  - contentId: String (unique)
  - subtopicId: ObjectId (ref: 'Subtopic', indexed)
  - version: Number
  - createdAt: Date
  
  - structure: {
      story: {
        text: String
        scenario: String
        problemStatement: String
        connectionToPrevious: String
      }
      
      conceptIntroduction: {
        text: String
        intuition: String
        realWorldAnalogy: String
      }
      
      theory: {
        mainContent: String // Rich text with formatting
        sections: [{
          title: String
          content: String
          order: Number
        }]
      }
      
      integration: {
        summary: String
        keyTakeaways: [String]
        connectionToNext: String
      }
    }
  
  - pedagogicalElements: [{
      type: Enum ['key_concept', 'example', 'pro_tip', 'warning', 'practice_problem', 'definition']
      title: String
      content: String
      position: Number // Where in theory content
      importance: Enum ['high', 'medium', 'low']
    }]
  
  - conceptsIntroduced: [String] // New concepts in this content
  - conceptsReinforced: [String] // Previously learned concepts referenced
  - terminology: [{
      term: String
      definition: String
      firstUsed: Boolean
    }]
  - notation: [{
      symbol: String
      meaning: String
      example: String
    }]
  
  - practiceProblems: [{
      problem: String
      difficulty: Enum ['easy', 'medium', 'hard']
      solution: String
      hints: [String]
    }]
  
  - metadata: {
      wordCount: Number
      readingTime: Number
      complexity: Number
      aiModel: String
      promptVersion: String
      generationTime: Number
    }
  
  Methods:
  - render()
  - update()
  - getNextContent()
```

#### **Context Memory Model**
```typescript
□ ContextMemorySchema
  Fields:
  - sessionId: ObjectId (ref: 'Session', unique, indexed)
  - updatedAt: Date
  
  - narrativeTimeline: [{
      subtopicId: ObjectId (ref: 'Subtopic')
      timestamp: Date
      storyElement: String
      problemSolved: String
      conceptsIntroduced: [String]
    }]
  
  - terminologyMap: Map<String, {
      definition: String
      firstIntroduced: ObjectId (ref: 'Subtopic')
      usageCount: Number
      aliases: [String]
    }>
  
  - notationMap: Map<String, {
      meaning: String
      firstIntroduced: ObjectId (ref: 'Subtopic')
      context: String
      relatedNotations: [String]
    }>
  
  - runningExamples: [{
      exampleId: String
      description: String
      introducedIn: ObjectId (ref: 'Subtopic')
      revisitedIn: [ObjectId] (ref: 'Subtopic')
      currentState: String
    }]
  
  - conceptDependencyGraph: {
      nodes: [{
        conceptId: String
        label: String
        subtopicId: ObjectId
      }]
      edges: [{
        from: String
        to: String
        relationship: Enum ['prerequisite', 'builds_on', 'related', 'contrasts']
      }]
    }
  
  - learningPath: [{
      subtopicId: ObjectId
      completedAt: Date
      understanding: Number // 1-10 score
    }]
  
  Methods:
  - addToTimeline()
  - getRecentContext()
  - addTerminology()
  - addNotation()
  - updateConceptGraph()
```

#### **Concept Graph Model**
```typescript
□ ConceptGraphSchema
  Fields:
  - sessionId: ObjectId (ref: 'Session', indexed)
  - subject: String
  
  - concepts: [{
      conceptId: String (unique within session)
      name: String
      definition: String
      category: String
      difficulty: Number
      subtopicId: ObjectId (ref: 'Subtopic')
    }]
  
  - relationships: [{
      fromConceptId: String
      toConceptId: String
      type: Enum ['prerequisite', 'extends', 'applies_to', 'similar_to', 'contrasts_with']
      strength: Number // 0-1
      description: String
    }]
  
  Methods:
  - addConcept()
  - addRelationship()
  - getPrerequisites()
  - getRelatedConcepts()
  - visualizeGraph()
```

#### **User Progress Model**
```typescript
□ UserProgressSchema
  Fields:
  - userId: String (indexed)
  - sessionId: ObjectId (ref: 'Session', indexed)
  - updatedAt: Date
  
  - completedSubtopics: [{
      subtopicId: ObjectId
      completedAt: Date
      timeSpent: Number
      understanding: Number // Self-reported 1-10
    }]
  
  - masteredConcepts: [{
      conceptId: String
      masteryLevel: Number // 0-100
      lastReviewed: Date
      practiceCount: Number
    }]
  
  - strugglingAreas: [{
      subtopicId: ObjectId
      difficulty: String
      attempts: Number
      lastAttempt: Date
    }]
  
  - statistics: {
      totalTimeSpent: Number
      averageSessionTime: Number
      completionRate: Number
      preferredLearningTime: String
    }
  
  Methods:
  - markSubtopicComplete()
  - updateMastery()
  - getRecommendations()
```

## **PHASE 3: API Endpoints**

### 3.1 Session Management Endpoints

```
□ POST /api/sessions
  Body: { subject, targetAudience, objectives }
  Actions:
  - Create session
  - Initialize context
  - Generate initial topics
  Response: { sessionId, topics }

□ GET /api/sessions/:sessionId
  Actions:
  - Get session details
  - Get progress
  Response: { session, progress, currentSubtopic }

□ GET /api/sessions/:sessionId/topics
  Actions:
  - Get all topics
  - Include subtopics
  Response: { topics }

□ POST /api/sessions/:sessionId/topics
  Body: { count }
  Actions:
  - Generate additional topics
  Response: { topics }

□ DELETE /api/sessions/:sessionId
  Actions:
  - Archive session
  - Clean up data
  Response: { success }
```

```
□ GET /api/topics/:topicId

  - Get topic details
  - Get subtopics

  - Generate subtopics

□ GET /api/topics/:topicId/prerequisites

  - Get prerequisite topics
```
