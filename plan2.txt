javascript/**
 * @file services/file/pdfExtractor.js
 * @description Extract text from PDF files
 */

// ============================================
// FUNCTION: extractTextFromPDF
// ============================================
/**
 * Extract text content from PDF
 * 
 * @param {Buffer|string} input - PDF buffer or file path
 * @returns {Promise<string>} extractedText
 * 
 * @workflow
 * 1. Load PDF using pdf-parse
 * 2. Extract text from all pages
 * 3. Clean extracted text:
 *    - Remove excessive whitespace
 *    - Fix line breaks
 *    - Handle tables and special characters
 * 4. Return clean text
 * 
 * @errors
 * - Throws if PDF is password protected
 * - Throws if PDF is corrupted
 * - Returns empty string if no text found
 */
exports.extractTextFromPDF = async (input) => {
  const pdfParse = require('pdf-parse');
  
  try {
    let dataBuffer;
    
    if (Buffer.isBuffer(input)) {
      dataBuffer = input;
    } else {
      // If file path provided
      const fs = require('fs').promises;
      dataBuffer = await fs.readFile(input);
    }
    
    const data = await pdfParse(dataBuffer);
    
    // Clean the text
    let text = data.text;
    text = text.replace(/\s+/g, ' '); // Normalize whitespace
    text = text.replace(/\n{3,}/g, '\n\n'); // Max 2 line breaks
    text = text.trim();
    
    return text;
  } catch (error) {
    throw new Error(`PDF extraction failed: ${error.message}`);
  }
};

/**
 * Get overall progress for a session
 * 
 * @param {number} sessionId
 * @returns {Promise<Object>} stats
 */
exports.getSessionProgress = async (sessionId) => {
  const { Subtopic, SubtopicProgress, Topic } = require('../../models');
  
  // Get all subtopics for session
  const subtopics = await Subtopic.findAll({
    include: [{
      model: Topic,
      where: { session_id: sessionId },
      attributes: ['id', 'name']
    }, {
      model: SubtopicProgress,
      required: false
    }]
  });
  
  const stats = {
    total: subtopics.length,
    notStarted: 0,
    introduction: 0,
    understanding: 0,
    mastering: 0,
    completed: 0,
    completionPercentage: 0
  };
  
  subtopics.forEach(st => {
    if (!st.progress) {
      stats.notStarted++;
    } else {
      stats[st.progress.current_step]++;
    }
  });
  
  stats.completionPercentage = Math.round(
    (stats.completed / stats.total) * 100
  );
  
  return stats;
};

  });
  
  const analytics = {
    completionMetrics: {
      totalSubtopics: 0,
      completed: 0,
      inProgress: 0,
      notStarted: 0,
      percentage: 0
    },
    timeMetrics: {
      totalTimeSpent: 0,
      averageTimePerStep: {},
      averageTimePerSubtopic: 0
    },
    learningVelocity: {
      subtopicsPerWeek: 0,
      estimatedCompletionDate: null
    },
    engagement: {
      totalQuestions: 0,
      questionsPerSubtopic: 0
    },
    strengths: [],
    needsImprovement: []
  };
  
  // Calculate metrics
  progressRecords.forEach(progress => {
    analytics.completionMetrics.totalSubtopics++;
    
    if (progress.current_step === 'completed') {
      analytics.completionMetrics.completed++;
      
      // Calculate time spent
      if (progress.started_at && progress.mastering_completed_at) {
        const timeSpent = new Date(progress.mastering_completed_at) - new Date(progress.started_at);
        analytics.timeMetrics.totalTimeSpent += timeSpent;
      }
    } else if (progress.current_step !== 'not_started') {
      analytics.completionMetrics.inProgress++;
    } else {
      analytics.completionMetrics.notStarted++;
    }
  });
  
  analytics.completionMetrics.percentage = Math.round(
    (analytics.completionMetrics.completed / analytics.completionMetrics.totalSubtopics) * 100
  );
  
  // Get question count
  const questionCount = await UserQuestion.count({
    where: { session_id: sessionId }
  });
  analytics.engagement.totalQuestions = questionCount;
  analytics.engagement.questionsPerSubtopic = 
    analytics.completionMetrics.completed > 0 
      ? (questionCount / analytics.completionMetrics.completed).toFixed(1)
      : 0;
  
  return analytics;
};

javascript/**
 * @file middleware/errorHandler.js
 */

const { AppError } = require('../utils/errors');
const logger = require('../utils/logger');

exports.errorHandler = (err, req, res, next) => {
  let error = { ...err };
  error.message = err.message;
  
  // Log error
  logger.error({
    message: err.message,
    stack: err.stack,
    url: req.originalUrl,
    method: req.method,
    body: req.body,
    user: req.user?.id
  });
  
  // Sequelize validation error
  if (err.name === 'SequelizeValidationError') {
    const message = err.errors.map(e => e.message).join(', ');
    error = new AppError(message, 400);
  }
  
  // Sequelize unique constraint error
  if (err.name === 'SequelizeUniqueConstraintError') {
    const message = 'Duplicate value entered';
    error = new AppError(message, 409);
  }
  
  // JWT errors
  if (err.name === 'JsonWebTokenError') {
    error = new AppError('Invalid token', 401);
  }
  
  if (err.name === 'TokenExpiredError') {
    error = new AppError('Token expired', 401);
  }
  
  // Send response
  res.status(error.statusCode || 500).json({
    success: false,
    error: {
      message: error.message || 'Internal server error',
      code: error.code || 'INTERNAL_ERROR',
      ...(process.env.NODE_ENV === 'development' && { stack: err.stack })
    }
  });
};

// Async error wrapper
exports.asyncHandler = (fn) => {
  return (req, res, next) => {
    Promise.resolve(fn(req, res, next)).catch(next);
  };
};

// ============================================
// File Upload Validation
// ============================================
exports.validateFileUpload = (req, res, next) => {
  if (!req.file) {
    throw new ValidationError('File is required');
  }
  
  const allowedTypes = ['application/pdf', 'application/vnd.openxmlformats-officedocument.wordprocessingml.document', 'image/jpeg', 'image/png'];
  
  if (!allowedTypes.includes(req.file.mimetype)) {
    throw new ValidationError('Invalid file type. Allowed: PDF, DOCX, JPG, PNG');
  }
  
  const maxSize = 10 * 1024 * 1024; // 10MB
  if (req.file.size > maxSize) {
    throw new ValidationError('File size exceeds 10MB limit');
  }
  
  next();
};

// ============================================
// Pagination Validation
// ============================================
exports.validatePagination = (req, res, next) => {
  const schema = Joi.object({
    page: Joi.number().integer().min(1).default(1),
    limit: Joi.number().integer().min(1).max(100).default(10),
    sortBy: Joi.string().default('createdAt'),
    order: Joi.string().valid('ASC', 'DESC').default('DESC')
  });
  
  const { error, value } = schema.validate(req.query);
  
  if (error) {
    throw new ValidationError(error.details[0].message);
  }
  
  req.pagination = value;
  next();
};
